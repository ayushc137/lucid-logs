DELETE goal_daily_stats;
LET $aggregated = (SELECT out AS goal_id, out.created_by AS created_by, time::floor(IF in.completed_at IS NOT NONE THEN in.completed_at ELSE in.start_date END, 1d) AS date, math::sum(quantity_value) AS daily_value, count() AS contribution_count, out.target.value AS target_value FROM task_goals WHERE out.deleted_at IS NONE GROUP BY goal_id, date);
FOR $row IN $aggregated { UPSERT goal_daily_stats SET goal_id = $row.goal_id, date = $row.date, created_by = $row.created_by, daily_value = $row.daily_value ?? 0, contribution_count = $row.contribution_count ?? 0, target_value = $row.target_value, status = IF $row.target_value IS NOT NONE AND ($row.daily_value ?? 0) >= $row.target_value THEN "met" ELSE "pending" END, updated_at = time::now() WHERE goal_id = $row.goal_id AND date = $row.date; };

-- Adds pre-aggregated table for goal progress tracking.
-- This migration creates:
--   1. goal_daily_stats: Pre-computed daily progress snapshots
--   2. Events: Automatic updates when task_goals change
--   3. Enhanced indexes for performance
--
-- Run: surreal sql --endpoint http://localhost:8000 --username root --password root --namespace lucid --database logs < db/migrations/004_goal_analytics.surql
--
-- =============================================================================
-- GOAL_DAILY_STATS TABLE (Pre-Aggregated Daily Progress)
-- =============================================================================
-- This table stores pre-computed daily goal statistics for O(1) lookups
-- and enables historical progress charts without expensive recalculations.
--
-- Benefits:
--   - O(1) lookup for any date's progress
--   - Enables historical charts and trends
--   - Supports time-range queries efficiently
DEFINE TABLE IF NOT EXISTS goal_daily_stats SCHEMAFULL PERMISSIONS FULL;
-- Core fields
DEFINE FIELD IF NOT EXISTS goal_id ON goal_daily_stats TYPE record<goals>;
DEFINE FIELD IF NOT EXISTS date ON goal_daily_stats TYPE datetime;
DEFINE FIELD IF NOT EXISTS created_by ON goal_daily_stats TYPE record<users>;
-- Progress metrics
DEFINE FIELD IF NOT EXISTS daily_value ON goal_daily_stats TYPE float DEFAULT 0;
DEFINE FIELD IF NOT EXISTS cumulative_value ON goal_daily_stats TYPE float DEFAULT 0;
DEFINE FIELD IF NOT EXISTS contribution_count ON goal_daily_stats TYPE int DEFAULT 0;
-- Snapshot state
DEFINE FIELD IF NOT EXISTS target_value ON goal_daily_stats TYPE option<float>;
DEFINE FIELD IF NOT EXISTS streak_at_date ON goal_daily_stats TYPE int DEFAULT 0;
DEFINE FIELD IF NOT EXISTS status ON goal_daily_stats TYPE string DEFAULT "pending";  -- "pending", "met", "missed", "exceeded"
-- Timestamps
DEFINE FIELD IF NOT EXISTS created_at ON goal_daily_stats TYPE datetime DEFAULT time::now();
DEFINE FIELD IF NOT EXISTS updated_at ON goal_daily_stats TYPE datetime DEFAULT time::now();
-- Indexes for efficient queries
DEFINE INDEX IF NOT EXISTS idx_goal_daily_unique ON goal_daily_stats COLUMNS goal_id, date UNIQUE;
DEFINE INDEX IF NOT EXISTS idx_goal_daily_user_date ON goal_daily_stats COLUMNS created_by, date;
DEFINE INDEX IF NOT EXISTS idx_goal_daily_date ON goal_daily_stats COLUMNS date;
-- =============================================================================
-- ENHANCED INDEXES FOR task_goals
-- =============================================================================
-- Add indexes to optimize period-based filtering on task_goals edges.
-- Index for faster goal progress lookups
DEFINE INDEX IF NOT EXISTS idx_task_goals_goal_completed ON task_goals COLUMNS out, in;
-- =============================================================================
-- EVENTS: Automatic stat updates on task_goals changes
-- =============================================================================
-- These events automatically update goal_daily_stats when task_goals are created,
-- updated, or deleted. This ensures stats are always up-to-date without manual intervention.
-- Event: When a task-goal link is created
-- =============================================================================
-- GOAL ANALYTICS OPTIMIZATION MIGRATION - SIMPLIFIED
-- =============================================================================
-- Creates goal_daily_stats table for O(1) progress lookups.
-- Table is SCHEMALESS for maximum compatibility with SurrealDB 2.6.
--
-- Note: Stats are updated via Go code in tasks/repository.go when task_goals
-- edges are created/updated. Database events were removed due to instability.
-- =============================================================================
-- Simple SCHEMALESS table for goal daily stats
DEFINE TABLE IF NOT EXISTS goal_daily_stats SCHEMALESS PERMISSIONS FULL;
-- Indexes for efficient queries
DEFINE INDEX IF NOT EXISTS idx_goal_daily_unique ON goal_daily_stats COLUMNS goal_id, date UNIQUE;
DEFINE INDEX IF NOT EXISTS idx_goal_daily_user_date ON goal_daily_stats COLUMNS created_by, date;
-- Enhanced index for task_goals lookups
DEFINE INDEX IF NOT EXISTS idx_task_goals_goal_completed ON task_goals COLUMNS out, in;
